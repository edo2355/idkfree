import React, { useState } from 'react';
import { View, Text, TextInput, TouchableOpacity, StyleSheet, Alert } from 'react-native';
import { useNavigation, useRoute } from '@react-navigation/native';
import { useSafeAreaInsets } from 'react-native-safe-area-context';
import { api } from '../api/client';

export function AddCatalogItemScreen() {
  const navigation = useNavigation<any>();
  const route = useRoute<any>();
  const itemToEdit = route.params?.item;
  const insets = useSafeAreaInsets();

  const [title, setTitle] = useState(itemToEdit ? itemToEdit.title : '');
  const [description, setDescription] = useState(itemToEdit ? itemToEdit.description : '');
  const [price, setPrice] = useState(itemToEdit ? String(itemToEdit.price) : '');
  const [type, setType] = useState(itemToEdit ? itemToEdit.type : 'product');
  const [category, setCategory] = useState(itemToEdit ? (itemToEdit.category || 'General') : 'General');

  async function handlePublish() {
    if (!title) return Alert.alert('ERROR', 'El título es obligatorio');
    try {
      if (itemToEdit) {
        await api.put(`/catalog/${itemToEdit.id}`, {
          title, description, price: parseFloat(price) || 0, type, category
        });
      } else {
        await api.post('/catalog', {
        title, description, price: parseFloat(price) || 0, type, currency: 'USD', category
        });
      }
      Alert.alert('SISTEMA', 'Item publicado correctamente');
      navigation.goBack();
    } catch (e: any) {
      Alert.alert('ERROR', e.message || 'No se pudo publicar');
    }
  }

  return (
    <View style={[styles.container, { paddingTop: insets.top }]}>
      <View style={styles.hud}>
        <Text style={styles.hudTitle}>{itemToEdit ? '[ EDITAR ITEM ]' : '[ NUEVO ITEM ]'}</Text>
        <TouchableOpacity onPress={() => navigation.goBack()}>
          <Text style={styles.hudClose}>[ X ]</Text>
        </TouchableOpacity>
      </View>

      <View style={styles.form}>
        <Text style={styles.label}>TÍTULO</Text>
        <TextInput style={styles.input} value={title} onChangeText={setTitle} placeholder="Ej. Camiseta IDKFREE" placeholderTextColor="#444" />

        <Text style={styles.label}>DESCRIPCIÓN</Text>
        <TextInput style={[styles.input, { height: 80 }]} value={description} onChangeText={setDescription} placeholder="Detalles del producto o servicio" placeholderTextColor="#444" multiline />

        <Text style={styles.label}>PRECIO (USD)</Text>
        <TextInput style={styles.input} value={price} onChangeText={setPrice} placeholder="0.00" placeholderTextColor="#444" keyboardType="numeric" />

        
        <Text style={styles.label}>CATEGORÍA (Ej. Ropa, Servicios)</Text>
        <TextInput style={styles.input} value={category} onChangeText={setCategory} placeholder="General" placeholderTextColor="#444" />

        <Text style={styles.label}>TIPO</Text>
        <View style={{ flexDirection: 'row', gap: 10, marginBottom: 20 }}>
          <TouchableOpacity style={[styles.typeBtn, type === 'product' && styles.typeBtnActive]} onPress={() => setType('product')}>
            <Text style={[styles.typeText, type === 'product' && { color: '#000' }]}>PRODUCTO</Text>
          </TouchableOpacity>
          <TouchableOpacity style={[styles.typeBtn, type === 'service' && styles.typeBtnActive]} onPress={() => setType('service')}>
            <Text style={[styles.typeText, type === 'service' && { color: '#000' }]}>SERVICIO</Text>
          </TouchableOpacity>
        </View>

        <TouchableOpacity style={styles.submitBtn} onPress={handlePublish}>
          <Text style={styles.submitText}>{itemToEdit ? 'GUARDAR CAMBIOS' : 'PUBLICAR ITEM'}</Text>
        </TouchableOpacity>
      </View>
    </View>
  );
}

// ---------- Helpers ----------
function hexFrom(str: string, len = 4): string {
  let h = 0;
  for (let i = 0; i < str.length; i++) h = (h * 31 + str.charCodeAt(i)) & 0xffffff;
  return h.toString(16).toUpperCase().padStart(6, '0').slice(0, len);
}

function makeBarcode(width: number, seed: string) {
  const bars: any[] = [];
  let s = 0;
  for (let i = 0; i < seed.length; i++) s = (s * 31 + seed.charCodeAt(i)) & 0xffffffff;
  const rand = () => {
    s = (s * 1103515245 + 12345) & 0x7fffffff;
    return s / 0x7fffffff;
  };
  let x = 0;
  while (x < width) {
    const w = Math.floor(rand() * 4) + 1;
    const gap = Math.floor(rand() * 3) + 1;
    bars.push({ x, w });
    x += w + gap;
  }
  return bars;
}

function BarcodeSVG({ width, height, color, seed }: any) {
  const bars = makeBarcode(width, seed);
  return (
    <Svg width={width} height={height} style={{ marginVertical: 6 }}>
      {bars.map((b: any, i: number) => (
        <Rect key={i} x={b.x} y={0} width={b.w} height={height} fill={color} />
      ))}
    </Svg>
  );
}

function WaveformSVG({ width, height, color }: any) {
  const path = (() => {
    let d = `M 0 ${height / 2}`;
    for (let x = 0; x <= width; x += 3) {
      const y = height / 2 + Math.sin(x / 7) * (height / 3) * Math.abs(Math.sin(x / 23));
      d += ` L ${x} ${y}`;
    }
    return d;
  })();
  return (
    <Svg width={width} height={height}>
      <Path d={path} stroke={color} strokeWidth={1} fill="none" />
    </Svg>
  );
}

function HexCluster({ size, color }: any) {
  const s = size;
  const hexPath = (cx: number, cy: number, r: number) => {
    let d = '';
    for (let i = 0; i < 6; i++) {
      const a = (Math.PI / 3) * i - Math.PI / 6;
      const x = cx + r * Math.cos(a);
      const y = cy + r * Math.sin(a);
      d += (i === 0 ? 'M' : 'L') + ` ${x} ${y} `;
    }
    return d + 'Z';
  };
  return (
    <Svg width={s * 4} height={s * 2}>
      {[0, 1, 2, 3].map((i) => (
        <Path
          key={i}
          d={hexPath(s / 2 + i * (s * 0.85), s / 2, s / 2 - 1)}
          stroke={color}
          strokeWidth={1}
          fill="none"
        />
      ))}
    </Svg>
  );
}

  const [sending, setSending] = useState(false);

  async function load() {
    try {
      const res = await api.messages.conversation(alias);
      setOther(res.other);
      setMessages(res.messages ?? []);
    } catch {}
    setLoading(false);
    setTimeout(() => listRef.current?.scrollToEnd({ animated: false }), 100);

  async function send() {
    if (!text.trim() || sending) return;

      await api.messages.send(alias, body);
      await load();

    setSending(false);

    const d = new Date(str.replace(' ', 'T') + 'Z');
    const s = Math.floor((Date.now() - d.getTime()) / 1000);

    if (s < 3600) return `${Math.floor(s / 60)}m`;
    if (s < 86400) return d.toLocaleTimeString('es-ES', { hour: '2-digit', minute: '2-digit' });
    return d.toLocaleDateString('es-ES', { day: '2-digit', month: '2-digit' });
  }

    return (
      <View style={{ flex: 1, backgroundColor: theme.bg, justifyContent: 'center', alignItems: 'center' }}>

<TouchableOpacity onPress={() => navigation.goBack()} activeOpacity={0.6} style={{ padding: 6 }}>

export function ConfigScreen({ onClose }: Props) {
  const [mode, setMode] = useState<Mode>('local');
  const [ip, setIp] = useState('127.0.0.1');
  const [customUrl, setCustomUrl] = useState('');

  const [savedUrl, setSavedUrl] = useState<string | null>(null);

  // Al montar: precargar la URL guardada y detectar si era local o remota
  useEffect(() => {
    (async () => {
      const stored = await AsyncStorage.getItem(STORAGE_KEY);
      if (!stored) return;
      setSavedUrl(stored);

      const isLocal =
        stored.startsWith('http://') &&
        !stored.includes('https://') &&
        (stored.includes('127.0.0.1') ||
          stored.includes('localhost') ||
          /^http:\/\/(10\.|192\.168\.|172\.)/.test(stored));

      if (isLocal) {
        setMode('local');
        const m = stored.match(/^http:\/\/([^\/]+)/);
        if (m) setIp(m[1].replace(/:3000$/, ''));
      } else {
        setMode('hosting');
        setCustomUrl(stored.replace(/\/api$/, ''));
      }
    })();
  }, []);

  function buildUrl(): string | null {
    if (mode === 'local') {
      const clean = ip.trim().replace(/^https?:\/\//, '').replace(/\/.*$/, '');
      if (!clean) return null;
      return `http://${clean}:3000/api`;
    } else {
      const url = customUrl.trim().replace(/\/$/, '');
      if (!url) return null;
      return url.endsWith('/api') ? url : `${url}/api`;
    }
  }

  async function testAndSave(): Promise<boolean> {
    const baseUrl = buildUrl();
    if (!baseUrl) {
      setMessage({ ok: false, text: 'Introduce una dirección válida' });
      return false;
    }
    setTesting(true);
    setMessage(null);

    const healthUrl = baseUrl.replace(/\/api$/, '/health');
    try {
      const controller = new AbortController();
      const timeout = setTimeout(() => controller.abort(), 60000);
      const res = await fetch(healthUrl, { signal: controller.signal });
      clearTimeout(timeout);

      if (res.ok) {
        await AsyncStorage.setItem(STORAGE_KEY, baseUrl);
        setMessage({ ok: true, text: '✓ Conectado correctamente' });
        setTimeout(onClose, 900);
        return true;
      } else {
        setMessage({ ok: false, text: `✗ Servidor respondió ${res.status}` });
        return false;
      }
    } catch (e: any) {
      const msg = e?.name === 'AbortError' ? 'Timeout (60s) - servidor dormido?' : 'No se pudo conectar';
      setMessage({ ok: false, text: `✗ ${msg}` });
      return false;
    }
    setTesting(false);
  }

  return (
    <View style={styles.overlay}>
    
      <ScreenBackground variant="grid" />
      <ScrollView contentContainerStyle={styles.card}>
        <Text style={styles.title}>Servidor</Text>
        <Text style={styles.subtitle}>

import { fonts } from '../theme/fonts';
import { api } from '../api/client';
import { CustomGalleryModal } from '../components/CustomGalleryModal';

type Mode = 'camera' | 'preview' | 'note';
type ShotType = 'photo' | 'video';
type TimerOpt = 0 | 3 | 10;
type VidQuality = '480p' | '720p' | '1080p' | '2160p';

const NOTE_COLORS = ['#0F0F0F', '#1A1A2E', '#2E1A1A', '#1A2E1A', '#2E2E1A', '#1A1A1A'];

export function CreateScreen() {
  const theme = useTheme();
  const navigation = useNavigation<any>();

  const [permission, requestPermission] = useCameraPermissions();
  const cameraRef = useRef<any>(null);
  const [facing, setFacing] = useState<'back' | 'front'>('back');
  const [flash, setFlash] = useState<'off' | 'on'>('off');
  const [shotType, setShotType] = useState<ShotType>('photo');
  const [isRecording, setIsRecording] = useState(false);
  const [recordingSeconds, setRecordingSeconds] = useState(0);

  // Native options
  const [timer, setTimer] = useState<TimerOpt>(0);
  const [timerCount, setTimerCount] = useState<number | null>(null);
  const [gridOn, setGridOn] = useState(false);
  const [mirrorOn, setMirrorOn] = useState(false);
  const [videoQuality, setVideoQuality] = useState<VidQuality>('1080p');
  const [autofocus, setAutofocus] = useState(true);
  const [settingsOpen, setSettingsOpen] = useState(false);
  const [galleryOpen, setGalleryOpen] = useState(false);

  const [mode, setMode] = useState<Mode>('camera');
  const [capturedUri, setCapturedUri] = useState<string | null>(null);
  const [uploadedUrl, setUploadedUrl] = useState<string | null>(null);
  const [uploading, setUploading] = useState(false);
  const [caption, setCaption] = useState('');
  const [posting, setPosting] = useState(false);

  const [noteText, setNoteText] = useState('');
  const [noteBg, setNoteBg] = useState(NOTE_COLORS[0]);

  useEffect(() => {
    let interval: any;
    if (isRecording) interval = setInterval(() => setRecordingSeconds(s => s + 1), 1000);
    else setRecordingSeconds(0);
    return () => clearInterval(interval);
  }, [isRecording]);

  if (!permission) return <View style={{ flex: 1, paddingBottom: 80, backgroundColor: '#000' }} />;
  if (!permission.granted) {
    return (
      <View style={{ flex: 1, backgroundColor: '#000', alignItems: 'center', justifyContent: 'center', padding: 24 }}>
        <Text style={{ color: '#FFF', fontFamily: 'monospace', marginBottom: 100, textAlign: 'center' }}>
          // SECURITY_ACCESS_REQUIRED{'\n\n'}Se requiere acceso a la cámara.
        </Text>
        <TouchableOpacity onPress={requestPermission} style={{ backgroundColor: '#32D3EE', paddingHorizontal: 20, paddingVertical: 12 }}>
          <Text style={{ color: '#000', fontFamily: 'monospace', fontWeight: 'bold', letterSpacing: 1 }}>CONCEDER ACCESO ◆</Text>
        </TouchableOpacity>
      </View>
    );
  }

  async function uploadMedia(uri: string, isVideo: boolean) {
    setUploading(true);
    try {
      const res = await api.uploads.upload({
        uri,
        name: isVideo ? `video-${Date.now()}.mp4` : `photo-${Date.now()}.jpg`,
        type: isVideo ? 'video/mp4' : 'image/jpeg',
      });
      setUploadedUrl(res.url);
    } catch (e: any) {
      Alert.alert('Error', e?.message ?? 'No se pudo subir');
      resetCamera();
    }
    setUploading(false);
  }

  async function takePhoto() {
    if (!cameraRef.current || timerCount !== null) return;

    // Timer countdown
    if (timer > 0) {
      for (let i = timer; i > 0; i--) {
        setTimerCount(i);
        await new Promise(r => setTimeout(r, 1000));
      }
      setTimerCount(null);
    }

    try {
      // CALIDAD MÁXIMA — sin compresión, sin procesamiento posterior
      const photo = await cameraRef.current.takePictureAsync({
        quality: 1,
        skipProcessing: false,
        exif: false,
      });
      setCapturedUri(photo.uri);
      setMode('preview');
      await uploadMedia(photo.uri, false);
    } catch (e: any) {
      setTimerCount(null);
      Alert.alert('Error', e?.message ?? 'No se pudo tomar la foto');
    }
  }

  async function startRecording() {
    if (!cameraRef.current || isRecording) return;
    try {
      setIsRecording(true);
      const video = await cameraRef.current.recordAsync({ maxDuration: 60 });
      setIsRecording(false);
      setCapturedUri(video.uri);
      setMode('preview');
      await uploadMedia(video.uri, true);
    } catch (e: any) {
      setIsRecording(false);
      Alert.alert('Error', e?.message ?? 'No se pudo grabar');
    }
  }

    if (cameraRef.current && isRecording) cameraRef.current.stopRecording();

    setShotType(isVideo ? 'video' : 'photo');

    await uploadMedia(uri, isVideo);

        visibility: 'followers',
      });
      navigation.navigate('Feed');
    } catch (e: any) {
      Alert.alert('Error', e?.message ?? 'No se pudo publicar');
    }
    setPosting(false);

      navigation.goBack();
    } catch (e: any) {
      Alert.alert('Error', e?.message ?? 'No se pudo publicar la nota');
    }
    setPosting(false);
  }

    setUploadedUrl(null);
    setCaption('');
    setMode('camera');

                >

export function EditProfileScreen() {
  const theme = useTheme();
  const navigation = useNavigation<any>();
  const { user } = useAuthStore();

  const [name, setName] = useState('');
  const [phoneNumber, setPhoneNumber] = useState(user?.phone_number || '');
  const [bio, setBio] = useState('');
  const [accountType, setAccountType] = useState('personal');
  const [category, setCategory] = useState('');
  const [links, setLinks] = useState<{title: string, url: string}[]>([]);
  
  const [avatar, setAvatar] = useState<string | null>(null);
  const [avatarPreview, setAvatarPreview] = useState<string | null>(null);
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [uploading, setUploading] = useState(false);

  useEffect(() => {
    (async () => {
      if (!user) return;
      try {
        const res = await api.profile.get(user.alias);
        setName(res.user.name ?? '');
        setBio(res.user.bio ?? '');
        setAvatar(res.user.avatar_url ?? null);
        setAccountType(res.user.account_type ?? 'personal');
        setCategory(res.user.category ?? '');
        
        if (res.user.links) {
          try {
            setLinks(JSON.parse(res.user.links));
          } catch {}
        }

        const resolved = await resolveAssetUrl(res.user.avatar_url);
        setAvatarPreview(resolved);
      } catch {}
      setLoading(false);
    })();
  }, [user?.alias]);

  async function pickAvatar() {
    const perm = await ImagePicker.requestMediaLibraryPermissionsAsync();
    if (!perm.granted) return Alert.alert('Permiso requerido', 'Necesitamos acceso a tu galería');
    const result = await ImagePicker.launchImageLibraryAsync({
      mediaTypes: ImagePicker.MediaTypeOptions.Images,
      quality: 0.85,
      allowsEditing: true,
      aspect: [1, 1],
    });
    if (result.canceled) return;
    const asset = result.assets[0];
    setAvatarPreview(asset.uri);
    setUploading(true);
    try {
      const res = await api.uploads.upload({
        uri: asset.uri,
        name: asset.fileName ?? `avatar-${Date.now()}.jpg`,
        type: asset.mimeType ?? 'image/jpeg',
      });
      setAvatar(res.url);
    } catch (e: any) {
      Alert.alert('Error', e?.message ?? 'No se pudo subir la imagen');
      const resolved = await resolveAssetUrl(avatar);
      setAvatarPreview(resolved);
    }
    setUploading(false);
  }

  async function save(): Promise<boolean> {
    setSaving(true);
    try {
      const validLinks = links.filter(l => l.title.trim() && l.url.trim());
      await api.profile.update({
        name: name.trim() || null,
        bio: bio.trim() || null,
        avatar_url: avatar,
        account_type: accountType,
        category: accountType === 'personal' ? null : category.trim(),
        links: validLinks.length > 0 ? JSON.stringify(validLinks) : null,
      });
      setSaving(false);
      navigation.goBack();
      return true;
    } catch (e: any) {
      Alert.alert('Error', e?.message ?? 'No se pudo guardar');
      setSaving(false);
      return false;
    }
  }

  function updateLink(index: number, key: 'title' | 'url', value: string) {
    const newLinks = [...links];
    if (!newLinks[index]) newLinks[index] = { title: '', url: '' };
    newLinks[index][key] = value;
    setLinks(newLinks);
  }

  if (loading) {

      <View style={{ flex: 1, backgroundColor: theme.bg, justifyContent: 'center', alignItems: 'center' }}>
        <ActivityIndicator color={theme.text} />
      </View>
    );

      <ScreenBackground variant="topo" />
      <ScrollView style={{ flex: 1 }} contentContainerStyle={{ padding: 20, paddingTop: 60, paddingBottom: 40 }}>

          <TouchableOpacity onPress={save} disabled={saving || uploading}>

              {avatarPreview ? <Image source={{ uri: avatarPreview }} style={{ width: '100%', height: '100%' }} /> : <Text style={{ color: theme.textMuted, fontSize: 32 }}>+</Text>}
              {uploading && <View style={{ position: 'absolute', top: 0, left: 0, right: 0, bottom: 0, backgroundColor: 'rgba(0,0,0,0.5)', alignItems: 'center', justifyContent: 'center' }}><ActivityIndicator color="#FFF" /></View>}
            </View>

import React, { useEffect, useState } from 'react';
import { View, Text, FlatList, TouchableOpacity, StyleSheet, Alert, Platform } from 'react-native';
import { useNavigation } from '@react-navigation/native';
import { useSafeAreaInsets } from 'react-native-safe-area-context';
import AsyncStorage from '@react-native-async-storage/async-storage';
import * as Print from 'expo-print';
import * as Sharing from 'expo-sharing';
import * as FileSystem from 'expo-file-system/legacy';
import * as IntentLauncher from 'expo-intent-launcher';

export function ErrorLogsScreen() {
  const navigation = useNavigation<any>();
  const insets = useSafeAreaInsets();
  const [logs, setLogs] = useState<any[]>([]);

  useEffect(() => {
    loadLogs();
  }, []);

  async function loadLogs() {
    try {
      // Asumimos que errorLogger.ts guarda los logs en AsyncStorage bajo esta clave
      const savedLogs = await AsyncStorage.getItem('@idkfree_error_logs');
      if (savedLogs) {
        setLogs(JSON.parse(savedLogs));
      } else {
        // Datos de prueba si no hay logs reales
        setLogs([
          { id: '1', time: new Date().toISOString(), message: 'Error de prueba: No autenticado', stack: '...' },
          { id: '2', time: new Date().toISOString(), message: 'Error de prueba: Network request failed', stack: '...' }
        ]);
      }
    } catch (e) {
      Alert.alert('ERROR', 'No se pudieron cargar los logs');
    }
  }

  async function clearLogs() {
    try {
      await AsyncStorage.removeItem('@idkfree_error_logs');
      setLogs([]);
      Alert.alert('SISTEMA', 'Logs eliminados');
    } catch (e) {}
  }


    async function generatePDF() {
    if (logs.length === 0) return Alert.alert('SISTEMA', 'No hay logs para exportar');

    try {
      const isSharingAvailable = await Sharing.isAvailableAsync();
      if (!isSharingAvailable) {
        Alert.alert('ERROR', 'Tu dispositivo no soporta la función de compartir archivos.');
        return;
      }

      const htmlContent = `
        <html>
          <body style="background-color: #0a0a0a; color: #ffffff; font-family: monospace; padding: 20px;">
            <h1 style="color: #00E5FF; text-align: center;">[ IDKFREE SYSTEM REPORT ]</h1>
            <p style="text-align: center; color: #888;">Fecha: ${new Date().toLocaleString()}</p>
            <hr style="border-color: #333; margin: 20px 0;" />
            ${logs.map(l => `
              <div style="margin-bottom: 15px; border-bottom: 1px solid #222; padding-bottom: 10px;">
                <p style="color: #ff0044; font-weight: bold; font-size: 14px;">> ${l.message}</p>
                <p style="color: #888; font-size: 12px;">TIMESTAMP: ${new Date(l.time).toLocaleString()}</p>
                <pre style="color: #666; font-size: 10px; white-space: pre-wrap; background-color: #111; padding: 10px; border: 1px solid #333;">${l.stack || 'No stack trace available'}</pre>
              </div>
            `).join('')}
            <p style="color: #666; font-size: 10px; margin-top: 50px; text-align: center;">[ FIN DEL REPORTE ]</p>
          </body>
        </html>
      `;

      // 1. Generar el PDF
      const { uri } = await Print.printToFileAsync({ html: htmlContent });

      if (Platform.OS === 'android') {
        // En Android, obtener la URI de contenido y abrir el PDF con una app externa
        const contentUri = await FileSystem.getContentUriAsync(uri);
        await IntentLauncher.startActivityAsync('android.intent.action.VIEW', {
          data: contentUri,
          flags: 1,
          type: 'application/pdf',
        });
      } else {
        // En iOS, usar Sharing directamente
        await Sharing.shareAsync(uri, { 
          mimeType: 'application/pdf', 
          dialogTitle: 'Compartir Reporte de Errores',
          UTI: '.pdf'
        });
      }

    } catch (error: any) {
      console.error('Error generando PDF:', error);
      Alert.alert('ERROR', `No se pudo generar el PDF: ${error.message || 'Error desconocido'}`);
    }
  }

  return (
    <View style={[styles.container, { paddingTop: insets.top }]}>
      <View style={styles.hud}>
        <Text style={styles.hudTitle}>[ LOGS DE ERROR ]</Text>
        <TouchableOpacity onPress={() => navigation.goBack()}>
          <Text style={styles.hudClose}>[ X ]</Text>
        </TouchableOpacity>

        quality: 0.6,
        base64: true,

      const asset = result.assets[0];
      const mime = asset.mimeType || 'image/jpeg';
      setter(`data:${mime};base64,${asset.base64}`);

import { useEffect, useState } from 'react';
import { View, Text, FlatList, TouchableOpacity, RefreshControl, ActivityIndicator } from 'react-native';

import { fonts } from '../theme/fonts';

import { Avatar } from '../components/Avatar';
import { HeaderBar } from '../components/HeaderBar';
import type { Notification } from '../../shared/types';
import { ScreenBackground } from '../components/ScreenBackground';

export function NotificationsScreen() {
  const theme = useTheme();
  const [items, setItems] = useState<Notification[]>([]);
  const [loading, setLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);

  async function load() {
    try {
      const res = await api.notifications.list({ limit: 50 });
      setItems(res.notifications ?? []);
    } catch {}
    setLoading(false);
  }

  async function handleRefresh() {
    setRefreshing(true);
    await load();
    setRefreshing(false);
  }

  async function markAllRead() {
    try { await api.notifications.readAll(); } catch {}
    setItems((prev) => prev.map((n) => ({ ...n, read_at: n.read_at ?? new Date().toISOString() })));
  }

  useEffect(() => { load(); }, []);

  function labelFor(n: Notification): string {
    switch (n.type) {
      case 'like': return 'le gustó tu post';
      case 'comment': return 'comentó tu post';
      case 'follow': return 'te siguió';
      case 'mention': return 'te mencionó';
      case 'repost': return 'reposteó tu post';
      default: return n.body ?? 'nueva actividad';
    }
  }

  return (
    <View style={{ flex: 1, backgroundColor: theme.bg }}>
      <ScreenBackground variant="pulse" />
      <HeaderBar showBack title="Notificaciones" />
      <View style={{ paddingHorizontal: 20, paddingBottom: 12, flexDirection: 'row', justifyContent: 'flex-end', alignItems: 'center' }}>
        {items.some((n) => !n.read_at) ? (
          <TouchableOpacity onPress={markAllRead} activeOpacity={0.6}>
            <Text style={{ color: theme.textMuted, fontFamily: fonts.medium, fontSize: 13 }}>
              Marcar leídas
            </Text>
          </TouchableOpacity>
        ) : null}
      </View>

      {loading ? (
        <View style={{ padding: 40, alignItems: 'center' }}>
          <ActivityIndicator color={theme.text} />
        </View>
      ) : (
        <FlatList
          data={items}
          keyExtractor={(n) => n.id}
          refreshControl={<RefreshControl refreshing={refreshing} onRefresh={handleRefresh} tintColor={theme.text} />}
          contentContainerStyle={{ paddingHorizontal: 16, paddingBottom: 40 }}
          ListEmptyComponent={
            <Text style={{ color: theme.textMuted, textAlign: 'center', paddingVertical: 60, fontFamily: fonts.regular }}>
              Sin notificaciones todavía
            </Text>
          }
          renderItem={({ item }) => (
            <View
              style={{
                flexDirection: 'row',
                alignItems: 'center',
                padding: 14,
                borderRadius: 10,
                marginBottom: 8,
                backgroundColor: item.read_at ? 'transparent' : theme.surface,
                borderWidth: 1,
                borderColor: item.read_at ? 'transparent' : theme.border,
              }}
            >
              <Avatar name={item.actor_name ?? item.actor_alias ?? '?'} url={item.actor_avatar} size={40} />
              <View style={{ flex: 1, marginLeft: 12 }}>
                <Text style={{ color: theme.text, fontFamily: fonts.regular, fontSize: 14, lineHeight: 19 }}>
                  <Text style={{ fontFamily: fonts.medium }}>{item.actor_name ?? item.actor_alias ?? 'Alguien'}</Text>
                  {' '}{labelFor(item)}

                  {getTimeAgo(new Date(item.created_at))}

function getTimeAgo(date: Date): string {
  const s = Math.floor((Date.now() - date.getTime()) / 1000);
  if (s < 60) return 'ahora';
  if (s < 3600) return `hace ${Math.floor(s / 60)} min`;
  if (s < 86400) return `hace ${Math.floor(s / 3600)} h`;
  if (s < 604800) return `hace ${Math.floor(s / 86400)} d`;
  return date.toLocaleDateString('es-ES');
}

import React, { useState, useEffect } from 'react';

import { useAuth } from '../stores/authStore';

export default function OfflineVenueScreen() {
  const [name, setName] = useState('');
  const [capacity, setCapacity] = useState('');
  const [unsynced, setUnsynced] = useState<any[]>([]);
  const { token } = useAuth();

  const cargarDatos = async () => {
    try {
      await initDb();
      const data = await getUnsyncedVenues();
      setUnsynced(data);
    } catch (error) {
      console.error(error);
      Alert.alert('Error DB', 'No se pudo inicializar la base de datos local.');
    }
  };

  useEffect(() => {
    cargarDatos();
  }, []);

  const handleSave = async () => {
    if (!name || !capacity) return Alert.alert('Error', 'Llena todos los campos');

    const venue = {
      id: Date.now().toString(),
      name,
      type: 'stadium',
      country: 'CU',
      city: 'La Habana',
      address: 'Dirección de prueba',
      capacity: parseInt(capacity),
      shape: 'rectangular',
      sections: [
        { name: 'General', layoutType: 'straight', price: 50, color: '#3B82F6', rows: [{ name: 'A', index: 1, seats: [{ number: 1, label: '1', x: 100, y: 100 }] }] }
      ]
    };

    try {
      await saveVenueOffline(venue);
      await cargarDatos();
      setName(''); setCapacity('');
      Alert.alert('Guardado', 'Recinto guardado localmente.');
    } catch (error) {
      Alert.alert('Error', 'No se pudo guardar el recinto.');
    }
  };

  const handleSync = async () => {
    if (!token) return Alert.alert('Error', 'No hay sesión activa');
    try {
      const count = await syncVenuesToCloud(token);
      await cargarDatos();
      Alert.alert('Sincronización', `Se sincronizaron ${count} recintos.`);
    } catch (error) {
      Alert.alert('Error', 'Fallo al sincronizar con la nube.');
    }
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Crear Recinto Offline</Text>
      <TextInput placeholder="Nombre del recinto" value={name} onChangeText={setName} style={styles.input} />
      <TextInput placeholder="Capacidad" value={capacity} onChangeText={setCapacity} keyboardType="numeric" style={styles.input} />
      <Button title="Guardar Localmente" onPress={handleSave} />

      <Text style={styles.subtitle}>Recintos sin sincronizar: {unsynced.length}</Text>
      <FlatList
        data={unsynced}
        keyExtractor={(item) => item.id}
        renderItem={({ item }) => <Text style={styles.item}>{item.name} - {item.capacity} cap.</Text>}

  const [profile, setProfile] = useState<any>(null);

  const [avatarUri, setAvatarUri] = useState<string | null>(null);
  const [tab, setTab] = useState<ProfileTab>('posts');

  async function load() {
    if (!user) return;
    setLoading(true);
    try {
      const res = await api.profile.get(user.alias);
      setProfile(res);
      if ((res?.user as any)?.accent_color) {
        useThemeStore.getState().setAccent((res.user as any).accent_color);
      }
      const [postsRes, vibesRes, legacyRes] = await Promise.all([
        api.posts.userPosts(res.user.id),
        api.vibes.userVibes(user.alias).catch(() => ({ vibes: [] })),
        api.vibes.legacy({ limit: 50 }).catch(() => ({ vibes: [] })),
      ]);
      setPosts(postsRes.posts ?? []);
      setVibes(vibesRes.vibes ?? []);
      setLegacyVibes((legacyRes.vibes ?? []).filter((v: any) => v.author_id === res.user.id));
    } catch {}
    setLoading(false);

    (async () => {

    })();

  if (loading || !profile) {
    return (

      Alert.alert('DATOS INCOMPLETOS', 'Faltan cardId, secret, acción o PIN.');

    }

      Alert.alert('FALTA CONTRASEÑA', 'Debes ingresar la nueva contraseña.');
      return false;

    try {
      if (action === 'panic') {

          [{ text: 'OK', onPress: () => navigation.navigate('Login') }]);

function ReelItem({ item, theme }: { item: any; theme: any }) {
  const { height } = useWindowDimensions();
  const [liked, setLiked] = useState(item.liked_by_me ?? false);
  const [likesCount, setLikesCount] = useState(item.likes_count ?? 0);

  const player = useVideoPlayer(item.video_url, (p) => {
    p.loop = true;
    p.muted = false;
  });

  async function toggleLike() {
    const nextLiked = !liked;
    setLiked(nextLiked);
    setLikesCount((prev: number) => nextLiked ? prev + 1 : prev - 1);
    try {
      if (nextLiked) await api.posts.like(item.id);
      else await api.posts.unlike(item.id);
    } catch {
      setLiked(!nextLiked);
    }
  }

  return (
    <View style={{ flex: 1, backgroundColor: '#000' }}>
      <VideoView player={player} style={StyleSheet.absoluteFill} contentFit="cover" nativeControls={false} />
      <View style={{ position: 'absolute', bottom: 0, left: 0, right: 0, height: 200, backgroundColor: 'rgba(0,0,0,0.4)' }} />

      <View style={{ position: 'absolute', right: 16, bottom: 20, alignItems: 'center', gap: 20 }}>

        setLoading(false);
      }
    }
    loadReels();

    async function loadStreams() {
      try {
        const res = await api.streams.active();
        setStreams(res.streams || res || []);

        setLoadingStreams(false);

import React, { useState, useEffect } from 'react';

import { CameraView, useCameraPermissions } from 'expo-camera';
import { useAuth } from '../stores/authStore'; // Asumiendo que tienes un hook de auth

export default function ScannerScreen() {
  const [permission, requestPermission] = useCameraPermissions();
  const [scanned, setScanned] = useState(false);
  const [scanResult, setScanResult] = useState<any>(null);
  const { token } = useAuth();

  if (!permission) return <View />;
  if (!permission.granted) {
    return (
      <View style={styles.container}>
        <Text style={{ textAlign: 'center', marginBottom: 20 }}>Necesitamos permiso de cámara para escanear tickets</Text>
        <Button onPress={requestPermission} title="Dar Permiso" />
      </View>
    );
  }

  const handleBarCodeScanned = async ({ data }: { data: string }) => {
    if (scanned) return;
    setScanned(true);

    try {
      const res = await fetch('https://idkfree.onrender.com/api/tickets/scan', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
        body: JSON.stringify({ qrPayload: data })
      });
      const result = await res.json();
      setScanResult(result);
      
      if (result.status === 'valid') {
        Alert.alert('✅ ACCESO PERMITIDO', `Asiento: ${result.ticket.section} - ${result.ticket.row}-${result.ticket.seat}`);
      } else if (result.status === 'used') {
        Alert.alert('⚠️ TICKET YA USADO', 'Este ticket ya fue escaneado.');
      } else {
        Alert.alert('❌ INVÁLIDO', 'El código QR no es válido.');
      }
    } catch (error) {
      Alert.alert('Error', 'No se pudo conectar con el servidor.');
    }

    setTimeout(() => { setScanned(false); setScanResult(null); }, 3000);
  };

  return (
    <View style={styles.container}>
      <CameraView
        style={StyleSheet.absoluteFill}
        barcodeScannerSettings={{ barcodeTypes: ['qr'] }}
        onBarcodeScanned={scanned ? undefined : handleBarCodeScanned}
      />
      <View style={styles.overlay}>
        <View style={styles.scanBox} />
        <Text style={styles.instructions}>Apunta el QR del ticket dentro del cuadro</Text>
      </View>
    </View>
  );
}

const TIPS = [
  'No lo sé, gratis o libre',
  'Aquí no tienes que saberlo todo',
  'Sin algoritmo, sin presión',
  'Tu cuenta, siempre tuya',
  'Personas reales, ideas reales',
  'More than a question',
  'Un lugar para pensar en voz alta',
  'Menos ruido, más sentido',
];

const BOOT_STEPS = [
  { at: 0,  txt: 'BOOTING IDKFREE OS v2.0.0' },
  { at: 15, txt: 'VERIFYING IDENTITY' },
  { at: 30, txt: 'LOADING ALDEA v1' },
  { at: 50, txt: 'SYNCING CARNET' },
  { at: 68, txt: 'DECRYPTING QRs' },
  { at: 90, txt: 'FINALIZING' },
];

const LOAD_MS = 4800;   // duración de la barra hasta llegar a 100%
const READY_MS = 1200;  // tiempo que se queda en READY antes de pasar

interface Props {
  onFinish: () => void;
}

export function SplashScreen({ onFinish }: Props) {
  const { width: SW } = useWindowDimensions();
  const [progress, setProgress] = useState(0);
  const [finished, setFinished] = useState(false);
  const [dotGreen, setDotGreen] = useState(false);
  const [tip, setTip] = useState(TIPS[0]);

  // Rotar tips
  useEffect(() => {
    setTip(TIPS[Math.floor(Math.random() * TIPS.length)]);
    const tipInterval = setInterval(() => {
      setTip(TIPS[Math.floor(Math.random() * TIPS.length)]);
    }, 1500);
    return () => clearInterval(tipInterval);
  }, []);

  // Progreso hasta 100%
  useEffect(() => {
    const tickMs = 50;
    const totalTicks = LOAD_MS / tickMs;
    let tick = 0;
    const progressInterval = setInterval(() => {
      tick++;
      const p = Math.min(100, Math.round((tick / totalTicks) * 100));
      setProgress(p);
      if (p >= 100) {
        clearInterval(progressInterval);
        setFinished(true);
      }
    }, tickMs);
    return () => clearInterval(progressInterval);
  }, []);

  // Secuencia final: punto azul → verde → onFinish
  useEffect(() => {
    if (!finished) return;
    const t1 = setTimeout(() => setDotGreen(true), 400);
    const t2 = setTimeout(() => onFinish(), 400 + READY_MS);
    return () => { clearTimeout(t1); clearTimeout(t2); };
  }, [finished, onFinish]);

  const visibleSteps = BOOT_STEPS.filter(s => progress >= s.at);
  const totalBlocks = 28;
  const filled = Math.round((progress / 100) * totalBlocks);
  const bar = '█'.repeat(filled) + '░'.repeat(totalBlocks - filled);
  const panelW = Math.min(SW - 48, 380);
  const dotColor = dotGreen ? '#22C55E' : '#3B82F6';

  return (
    <View style={{ flex: 1, backgroundColor: darkTheme.bg, alignItems: 'center', justifyContent: 'center', padding: 24 }}>

      <ScreenBackground variant="particles" />
      <CarnetTicketsBackground />

      {/* Logo + título */}
      <Image
        source={require('../../assets/icon.png')}
        style={{ width: 84, height: 84, marginBottom: 12 }}
        resizeMode="contain"
      />
      <Text
        style={{
          fontSize: 30,
          fontFamily: fonts.displayBoldItalic,
          color: darkTheme.text,
          letterSpacing: 1.5,
          marginBottom: 6,
        }}
      >
        IDKFREE
      </Text>
      <Text
        style={{
          fontSize: 10,
          fontFamily: 'monospace',
          color: darkTheme.textSubtle,
          letterSpacing: 3,
          marginBottom: 28,
        }}
      >
        [ SYSTEM BOOT · v2.0.0 ]
      </Text>

      {/* Panel de boot */}
      <View
        style={{
          width: panelW,
          borderWidth: 1,
          borderColor: finished ? '#22C55E' : darkTheme.border,
          backgroundColor: 'rgba(15,15,15,0.85)',
          padding: 16,
          marginBottom: 20,
        }}
      >
        {/* Steps */}
        <View style={{ minHeight: BOOT_STEPS.length * 18 }}>
          {visibleSteps.map((s, i) => (
            <View key={i} style={{ flexDirection: 'row', alignItems: 'center', marginBottom: 2 }}>
              <Text style={{ fontSize: 10, fontFamily: 'monospace', color: darkTheme.textSubtle, marginRight: 6 }}>&gt;</Text>
              <Text style={{ fontSize: 10, fontFamily: 'monospace', color: darkTheme.text, letterSpacing: 0.5, flex: 1 }}>{s.txt}</Text>
              <Text style={{ fontSize: 10, fontFamily: 'monospace', color: '#22C55E', letterSpacing: 1 }}>[OK]</Text>
            </View>
          ))}
        </View>

        {/* Barra */}
        <View style={{ marginTop: 12 }}>
          <Text
            style={{
              fontSize: 12,
              fontFamily: 'monospace',
              color: finished ? '#22C55E' : darkTheme.text,
              letterSpacing: 0.5,
              marginBottom: 6,
            }}
          >
            {bar}
          </Text>
          <View style={{ flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center' }}>
            <Text style={{ fontSize: 9, fontFamily: 'monospace', color: darkTheme.textSubtle, letterSpacing: 2 }}>

  async function handleStart() {
    if (!title.trim()) return;
    setLoading(true);
    try {
      // 1. Le pedimos al backend que cree la sala del stream
      const res = await api.streams.start(title.trim());
      
      // 2. Navegamos a la pantalla de la cámara con el ID de la sala

        roomName: res.id || res.stream?.id, 
        isHost: true

    } finally {
      setLoading(false);
    }

  return (
    <View style={{ flex: 1, backgroundColor: theme.bg, padding: 20, justifyContent: 'center' }}>

  const navigation = useNavigation<any>();
  const alias: string = route.params?.alias;
  const [profile, setProfile] = useState<any>(null);
  const [posts, setPosts] = useState<any[]>([]);
  const [vibes, setVibes] = useState<any[]>([]);
  const [legacyVibes, setLegacyVibes] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);
  const [busy, setBusy] = useState(false);
  const [avatarUri, setAvatarUri] = useState<string | null>(null);
  const [tab, setTab] = useState<ProfileTab>('posts');

  async function load() {
    setLoading(true);
    try {
      const res = await api.profile.get(alias);
      setProfile(res);
      const [postsRes, vibesRes, legacyRes] = await Promise.all([
        api.posts.userPosts(res.user.id),
        api.vibes.userVibes(alias).catch(() => ({ vibes: [] })),
        api.vibes.legacy({ limit: 50 }).catch(() => ({ vibes: [] })),
      ]);
      setPosts(postsRes.posts ?? []);
      setVibes(vibesRes.vibes ?? []);
      setLegacyVibes((legacyRes.vibes ?? []).filter((v: any) => v.author_id === res.user.id));
    } catch (e) { console.log('profile error', e); }
    setLoading(false);
  }

  useFocusEffect(useCallback(() => { load(); }, [alias]));

  useEffect(() => {
    (async () => {
      if (!profile?.user) { setAvatarUri(null); return; }
      setAvatarUri(await resolveAssetUrl(profile.user.avatar_url));
    })();
  }, [profile]);

  async function toggleFollow() {
    if (!profile || busy || profile.is_me) return;
    setBusy(true);
    const wasFollowing = profile.is_following;
    try {
      if (wasFollowing) await api.follows.unfollow(profile.user.id);
      else await api.follows.follow(profile.user.id);
      setProfile({
        ...profile,
        is_following: !wasFollowing,
        stats: {
          ...profile.stats,
          followers: wasFollowing ? profile.stats.followers - 1 : profile.stats.followers + 1,
        },
      });
    } catch {}
    setBusy(false);
  }

  if (loading || !profile) {
    return (
      <View style={{ flex: 1, backgroundColor: theme.bg, justifyContent: 'center', alignItems: 'center' }}>
        <ActivityIndicator color={theme.text} />
      </View>
    );
  }

  const u = profile.user;
  const stats = profile.stats;
  const photoPosts = posts.filter((p: any) => p.image_url || p.video_url);
  const textPosts = posts.filter((p: any) => !p.image_url && !p.video_url);

  return (
    <ScrollView style={{ flex: 1, backgroundColor: theme.bg }} contentContainerStyle={{ paddingBottom: 40 }}>
      {/* Back */}
      <View style={{ paddingTop: 60, paddingHorizontal: 16, paddingBottom: 8 }}>
        <TouchableOpacity onPress={() => navigation.goBack()} activeOpacity={0.6}>
          <Text style={{ color: theme.text, fontFamily: fonts.medium, fontSize: 15 }}>← Atrás</Text>
        </TouchableOpacity>
      </View>

      {/* Avatar + info */}
      <View style={{ alignItems: 'center', marginTop: 12 }}>
        {avatarUri ? (
          <Image source={{ uri: avatarUri }} style={{ width: 96, height: 96, borderRadius: 48 }} />
        ) : (
          <View style={{ width: 96, height: 96, borderRadius: 48, backgroundColor: theme.surfaceElevated, justifyContent: 'center', alignItems: 'center' }}>
            <Text style={{ color: theme.text, fontSize: 36, fontFamily: fonts.displayBoldItalic }}>
              {(u.name ?? u.alias).charAt(0).toUpperCase()}
            </Text>
          </View>
        )}
        <Text style={{ color: theme.text, fontSize: 22, marginTop: 12, fontFamily: fonts.medium }}>
          {u.name ?? u.alias}
        </Text>
        <Text style={{ color: theme.textMuted, fontSize: 14, fontFamily: fonts.regular }}>
          @{u.alias}
        </Text>
        {u.bio ? (
          <Text style={{ color: theme.text, fontSize: 14, marginTop: 10, textAlign: 'center', paddingHorizontal: 32, fontFamily: fonts.regular, lineHeight: 20 }}>
            {u.bio}
          </Text>
        ) : null}
      </View>

      {/* Stats Hero OSHIMA */}
      <StatsHero
        stats={stats}
        joinedAt={u.created_at ? String(u.created_at).slice(0, 10).replace(/-/g, '.') : undefined}
      />

      {/* Botón seguir */}
      {!profile.is_me ? (
        <View style={{ paddingHorizontal: 32, marginTop: 20, marginBottom: 16 }}>
          <TouchableOpacity
            onPress={toggleFollow}
            disabled={busy}
            activeOpacity={0.7}
            style={{
              paddingVertical: 12, borderRadius: 8, alignItems: 'center',
              backgroundColor: profile.is_following ? 'transparent' : theme.text,
              borderWidth: profile.is_following ? 1 : 0,
              borderColor: theme.border,
            }}
          >
            <Text style={{ color: profile.is_following ? theme.text : theme.bg, fontFamily: fonts.medium, fontSize: 14 }}>
              {profile.is_following ? 'Siguiendo' : 'Seguir'}
            </Text>
          </TouchableOpacity>
        </View>
      ) : null}

      {/* Tabs */}
      <ProfileTabs active={tab} onChange={setTab} />

      {/* Contenido */}
      <View style={{ padding: 2 }}>
        {tab === 'posts' ? (
          <PostGrid posts={photoPosts.length > 0 ? photoPosts : posts} />
        ) : tab === 'reels' ? (
          <View style={{ padding: 16 }}>
            {textPosts.length === 0 && vibes.length === 0 ? (
              <Text style={{ color: theme.textMuted, textAlign: 'center', paddingVertical: 40, fontFamily: fonts.regular }}>
                Sin notas
              </Text>
            ) : (
              <>
                {textPosts.map((p) => (
                  <View key={p.id} style={{ backgroundColor: theme.surface, borderWidth: 1, borderColor: theme.border, borderRadius: 12, padding: 14, marginBottom: 10 }}>
                    <Text style={{ color: theme.text, fontFamily: fonts.regular, fontSize: 14, lineHeight: 20 }}>
                      {p.body}
                    </Text>
                  </View>
                ))}
                {vibes.map((v) => (
                  <View key={v.id} style={{ backgroundColor: theme.surface, borderWidth: 1, borderColor: theme.border, borderRadius: 12, padding: 14, marginBottom: 10 }}>
                    <Text style={{ color: theme.text, fontFamily: fonts.regular, fontSize: 14, lineHeight: 20 }}>
                      {v.message}
                    </Text>
                  </View>
                ))}
              </>
            )}
          </View>
        ) : (
          <View style={{ padding: 16 }}>
            {legacyVibes.length === 0 ? (
              <View style={{ paddingVertical: 40, alignItems: 'center' }}>
                <Text style={{ fontSize: 40, marginBottom: 10 }}>[LEGACY]️</Text>
                <Text style={{ color: theme.textMuted, textAlign: 'center', fontFamily: fonts.regular, fontSize: 13 }}>
                  Sin contenido legacy
                </Text>
              </View>
            ) : (
              legacyVibes.map((v) => (
                <View key={v.id} style={{ backgroundColor: theme.surface, borderWidth: 1, borderColor: theme.border, borderRadius: 12, padding: 14, marginBottom: 10 }}>
                  <Text style={{ color: theme.text, fontFamily: fonts.regular, fontSize: 14, lineHeight: 20 }}>
                    {v.message}
                  </Text>
                </View>
              ))
            )}
          </View>
        )}
      </View>
    </ScrollView>
  );
}

  tranquilo: '[WAVE]',
  nose: '[?]',
};

  const [loading, setLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);

  async function load() {
    setLoading(true);
    try {
      let res;
      if (tab === 'legado') {
        res = await api.vibes.legacy({ limit: 50 });
      } else if (tab === 'todos') {
        res = await api.vibes.all({ limit: 50 });
      } else {
        res = await api.vibes.list({ limit: 50 });
      }
      setVibes(res.vibes ?? []);
    } catch { setVibes([]); }
    setLoading(false);
  }

  useEffect(() => { load(); }, [tab]);

  async function onRefresh() {
    setRefreshing(true);
    await load();
    setRefreshing(false);

  return (
